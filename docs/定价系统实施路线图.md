```sql
-- 在 Supabase SQL Editor 中执行
CREATE TABLE IF NOT EXISTS public.users (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  email TEXT NOT NULL,
  tier TEXT NOT NULL DEFAULT 'free' CHECK (tier IN ('guest', 'free', 'member')),
  member_until TIMESTAMPTZ,
  remaining_pdf_exports INTEGER NOT NULL DEFAULT 0,
  remaining_png_exports INTEGER NOT NULL DEFAULT 0,
  remaining_ats_checks INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 自动更新 updated_at 的触发器
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_users_updated_at
  BEFORE UPDATE ON public.users
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- 注册时自动创建用户记录的触发器
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.users (id, email)
  VALUES (NEW.id, NEW.email);
  RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- 启用 RLS
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;

-- 用户只能读取自己的数据
CREATE POLICY "Users can read own data"
  ON public.users FOR SELECT
  USING (auth.uid() = id);

-- 仅服务端可写入
CREATE POLICY "Only server can update users"
  ON public.users FOR ALL
  USING (auth.uid() = id)
  WITH CHECK (auth.uid() = id);
```

### 3.4 P1-3：会员过期检查

```typescript
// 会员过期检查（login 时自动执行）
function checkMembershipExpiry(userData: any): { tier: string; memberUntil: string | null } {
  if (userData.tier !== 'member') {
    return { tier: userData.tier, memberUntil: userData.member_until };
  }
  
  if (userData.member_until && new Date(userData.member_until) < new Date()) {
    // 会员已过期，自动降级
    return { tier: 'free', memberUntil: null };
  }
  
  return { tier: 'member', memberUntil: userData.member_until };
}
```

### 4.6 P2-5：支付状态轮询 Hook

```typescript
function usePaymentPolling(orderId: string | null) {
  const [status, setStatus] = useState<'pending' | 'paid' | 'timeout'>('pending');
  
  useEffect(() => {
    if (!orderId) return;
    
    const INTERVAL = 2000;   // 2s
    const TIMEOUT = 300000;  // 5min
    const startTime = Date.now();
    
    const interval = setInterval(async () => {
      if (Date.now() - startTime > TIMEOUT) {
        clearInterval(interval);
        setStatus('timeout');
        return;
      }
      
      const res = await fetch(`/api/payment/query/${orderId}`);
      const data = await res.json();
      if (data.status === 'completed') {
        clearInterval(interval);
        setStatus('paid');
      }
    }, INTERVAL);
    
    return () => clearInterval(interval);
  }, [orderId]);
  
  return status;
}
```
